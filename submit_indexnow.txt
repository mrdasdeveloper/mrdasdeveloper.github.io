#!/usr/bin/env python3
"""
Submit indexable URLs to IndexNow (Bing, Yandex, Seznam, Naver).

Setup (one-time):
  1. Run: python3 submit_indexnow.py --init
  2. Deploy {KEY}.txt to https://www.rameshdas.dev/{KEY}.txt (committed to repo)
  3. Configure the same key in Ahrefs Site Audit → IndexNow settings

Usage:
  python3 submit_indexnow.py --dry-run          # preview URL count
  python3 submit_indexnow.py                    # submit all urls.txt entries
  python3 submit_indexnow.py --limit 100        # submit first 100 only
  python3 submit_indexnow.py --removed URL ...  # notify removals
"""

from __future__ import annotations

import argparse
import json
import secrets
import sys
import urllib.error
import urllib.request
from pathlib import Path

ROOT = Path(__file__).resolve().parent
BASE_URL = "https://www.rameshdas.dev"
HOST = "www.rameshdas.dev"
CONFIG_PATH = ROOT / "indexnow_config.json"
URLS_PATH = ROOT / "urls.txt"
INDEXNOW_ENDPOINT = "https://api.indexnow.org/indexnow"
BATCH_SIZE = 10_000


def load_config() -> dict:
  if not CONFIG_PATH.is_file():
    raise SystemExit(
      "IndexNow not initialized. Run: python3 submit_indexnow.py --init"
    )
  return json.loads(CONFIG_PATH.read_text(encoding="utf-8"))


def init_key() -> None:
  key = secrets.token_hex(16)
  key_file = ROOT / f"{key}.txt"
  key_file.write_text(key + "\n", encoding="utf-8")
  config = {
    "key": key,
    "host": HOST,
    "keyLocation": f"{BASE_URL}/{key}.txt",
  }
  CONFIG_PATH.write_text(json.dumps(config, indent=2) + "\n", encoding="utf-8")
  print(f"✅  Created key file: {key_file.name}")
  print(f"✅  Wrote config: {CONFIG_PATH.name}")
  print(f"    keyLocation: {config['keyLocation']}")
  print("\nDeploy to GitHub Pages, then use this key in Ahrefs IndexNow settings.")


def load_urls(limit: int | None = None) -> list[str]:
  if not URLS_PATH.is_file():
    raise SystemExit(f"Missing {URLS_PATH.name}. Run: python3 update_sitemap.py && python3 generate_urls_txt.py")
  urls: list[str] = []
  for line in URLS_PATH.read_text(encoding="utf-8").splitlines():
    line = line.strip()
    if not line or line.startswith("#"):
      continue
    urls.append(line)
  if limit:
    urls = urls[:limit]
  return urls


def post_indexnow(config: dict, url_list: list[str]) -> None:
  payload = json.dumps(
    {
      "host": config["host"],
      "key": config["key"],
      "keyLocation": config["keyLocation"],
      "urlList": url_list,
    }
  ).encode("utf-8")
  req = urllib.request.Request(
    INDEXNOW_ENDPOINT,
    data=payload,
    headers={"Content-Type": "application/json; charset=utf-8"},
    method="POST",
  )
  try:
    with urllib.request.urlopen(req, timeout=60) as resp:
      print(f"  ✓  Batch of {len(url_list)} URLs → HTTP {resp.status}")
  except urllib.error.HTTPError as e:
    body = e.read().decode("utf-8", errors="replace")[:500]
    raise SystemExit(f"IndexNow HTTP {e.code}: {body}") from e


def main() -> int:
  parser = argparse.ArgumentParser(description="Submit URLs to IndexNow")
  parser.add_argument("--init", action="store_true", help="Generate IndexNow API key + key file")
  parser.add_argument("--dry-run", action="store_true", help="Print URLs without submitting")
  parser.add_argument("--limit", type=int, default=0, help="Max URLs to submit this run")
  parser.add_argument("--removed", nargs="*", default=[], help="URLs removed from site (optional)")
  args = parser.parse_args()

  if args.init:
    init_key()
    return 0

  config = load_config()
  limit = args.limit if args.limit > 0 else None
  urls = load_urls(limit)

  if args.removed:
    urls = list(dict.fromkeys(args.removed + urls))

  print(f"IndexNow — {len(urls)} URL(s) from {URLS_PATH.name}")
  print(f"  keyLocation: {config['keyLocation']}")

  if args.dry_run:
    for u in urls[:10]:
      print(f"    {u}")
    if len(urls) > 10:
      print(f"    … and {len(urls) - 10} more")
    return 0

  for i in range(0, len(urls), BATCH_SIZE):
    batch = urls[i : i + BATCH_SIZE]
    post_indexnow(config, batch)

  print(f"\n✅  Submitted {len(urls)} URL(s) to IndexNow")
  return 0


if __name__ == "__main__":
  raise SystemExit(main())
